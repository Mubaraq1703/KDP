% Chapter opener — one decorative title, one fresh page, one TOC entry, and no duplicate chapter heading.
\newcommand{\chapteropener}[2]{%
  \clearpage%
  \thispagestyle{plain}%
  \markboth{#2}{#2}%
  \addcontentsline{toc}{chapter}{#1: #2}%
  \vspace*{30pt}%
  \begin{center}
    {\Large\bfseries\color{accent}\MakeUppercase{#1}}\\[8pt]
    {\color{accent}\rule{0.3\textwidth}{0.5pt}}\\[10pt]
    {\Huge\bfseries\color{textgray} #2}\\[10pt]
    {\color{accent}\rule{0.3\textwidth}{0.5pt}}
  \end{center}
  \vspace{16pt}%
}

% Recipe box — styled block with tinted background
% Uses a tikzpicture node for the box, avoiding minipage list issues.
% CRITICAL: the node body is opened with \bgroup and closed with \egroup
% (NOT { and }). Classic \newenvironment stores begin/end code in separate
% macros; a {....} pair split across them is an unbalanced-brace error that
% nonstopmode "tolerates" by DROPPING the box content from the PDF.
\newenvironment{recipebox}[1]{%
  \par\vspace{8pt}%
  \noindent%
  \begin{tikzpicture}[remember picture]
    \node[anchor=north west, inner sep=6pt, text width=\dimexpr\textwidth-8pt, fill=recipedbg, draw=accent, line width=1.5pt, rounded corners=2pt] (box) \bgroup%
      \vspace{2pt}%
      {\Large\bfseries\color{accent}#1}\\[4pt]
      {\color{rulegray}\rule{\linewidth}{0.3pt}}\\[4pt]%
}{%
      \vspace{2pt}%
    \egroup;
  \end{tikzpicture}%
  \par\vspace{8pt}%
}

% Callout / highlight box — for tips, key concepts, comparisons.
% Same safe \bgroup/\egroup pattern as recipebox. Copy this environment
% (rename it) when a book needs its own styled callout — do NOT invent a
% new tikz box with { } braces.
\newenvironment{accentbox}[1]{%
  \par\vspace{8pt}%
  \noindent%
  \begin{tikzpicture}
    \node[inner sep=8pt, text width=\dimexpr\textwidth-16pt\relax, fill=accentlight, draw=accent, line width=1.5pt, rounded corners=2pt] \bgroup%
      \vspace{2pt}%
      {\Large\bfseries\color{accent}#1}\\[4pt]
      {\color{rulegray}\rule{\linewidth}{0.3pt}}\\[4pt]%
}{%
      \vspace{2pt}%
    \egroup;%
  \end{tikzpicture}%
  \par\vspace{8pt}%
}

% ---------- TABLE RULES (margin safety on A5) ----------
% Inside ANY box environment (recipebox, accentbox) the text column is
% \linewidth, which is NARROWER than \textwidth. Tables in boxes must use
% \begin{tabularx}{\linewidth}{...} — never \textwidth (they bleed past
% the page margin).
% For plain tabular with fixed p{...} columns, remember the padding:
%   total width = sum of p-widths + 2 * (number of columns) * \tabcolsep
%   (\tabcolsep defaults to 6pt per side). If the total exceeds
%   \linewidth the table bleeds out of the margin. Prefer tabularx:
%   \begin{tabularx}{\linewidth}{p{0.3\linewidth}p{0.2\linewidth}X}
% where X columns absorb the remaining width automatically.

% Yield line
\newcommand{\yieldline}[1]{%
  \begin{center}
    \vspace{4pt}
    {\small\itshape\color{rulegray}#1}
    \vspace{6pt}
  \end{center}
}
